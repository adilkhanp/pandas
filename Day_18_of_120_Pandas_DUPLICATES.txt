import pandas as pd

employees = pd.read_csv("D:\employees.csv")

print(employees.to_string())

print(employees[employees.duplicated("FIRST_NAME")].to_string())

print(employees[employees.duplicated(
    subset= ["FIRST_NAME" , "LAST_NAME"]
)].to_string())

print(employees.drop_duplicates(
    "FIRST_NAME"
)["FIRST_NAME"].to_string())

print(employees.drop_duplicates(
    subset = ["FIRST_NAME","LAST_NAME"]
)["FIRST_NAME"].to_string())

print( employees.groupby("FIRST_NAME")["FIRST_NAME"].count().to_string())
