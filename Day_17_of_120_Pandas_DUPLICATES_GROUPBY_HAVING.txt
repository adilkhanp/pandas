################ groupBy ###################

# syntax : dataframe.groupby(columnslist)[column_name].function
# syntax : dataframe.groupby(columnlist).agg( aliasname = ( columnname , function),
#                                                aliasname = ( columnname , function ),
#                                                 .
#                                                 .

 print(employees.groupby("DEPARTMENT_ID")["SALARY"].sum()) #select department_id , sum(salary) from employees group by department_id;
 print(employees.groupby("DEPARTMENT_ID")["SALARY"].max()) #select department_id , max(salary) from employees group by department_id;

 print(employees.groupby("DEPARTMENT_ID").agg(
         TotalSalary = ("SALARY",'sum'),
         MaximumSalary = ("SALARY",'max'),
         MinumumSalary = ("SALARY",'min'),
         commission_count = ("COMMISSION_PCT",'count')
 ))
