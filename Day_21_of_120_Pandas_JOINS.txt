import pandas as pd

employees = pd.read_csv("D:\employees.csv")
departments = pd.read_csv("D:\departments.csv")
locations = pd.read_csv("D:\locations.csv")

print( employees.merge(departments,
                       on = "DEPARTMENT_ID",
                       how = "inner",
                       suffixes= ( "_emp" , "_dept"))\
                # .get(["DEPARTMENT_NAME","FIRST_NAME" , "SALARY"])
                # .to_string()
                .columns
     )

print( employees.merge(departments,
                       on = "DEPARTMENT_ID",
                       how = "inner")
                 .groupby("DEPARTMENT_NAME").count()
                 .to_string() )

print( employees.merge(departments,
                       on = "DEPARTMENT_ID",
                       how = "inner")
                 .groupby("DEPARTMENT_NAME").size())

print( employees.merge(departments,
                       on = "DEPARTMENT_ID",
                       how = "inner")
                 .groupby("DEPARTMENT_NAME")["SALARY"].agg([  "min", "max" , "mean" , "count" ]))

# MIN , MAX , COUNT OF COMMISSION EMPLOYEES

print( employees.merge(departments,
                       on = "DEPARTMENT_ID",
                       how = "inner")
                 .groupby("DEPARTMENT_NAME").agg( Maximum_salary = ("SALARY", "max"),
                                                  Minimum_salary = ("SALARY", "min"),
                                                  Commission_emp = ("COMMISSION_PCT","count")))

print( employees.merge(departments,
                       on = "DEPARTMENT_ID",

       )
